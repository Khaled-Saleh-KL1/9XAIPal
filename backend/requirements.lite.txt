# Runtime dependencies for anything that never runs MinerU: the api service
# (Dockerfile.lite — PDF extraction is celery_worker's job alone, via
# Dockerfile.mineru + the full requirements.txt) and CI's test suite.
#
# This is requirements.txt MINUS `mineru[core]`.
#
# Why: MinerU is never imported by app/ — it is invoked as a CLI subprocess
# (see `mineru_binary` in app/core/config.py), by celery_worker alone.
# Installing it anyway drags in torch/OpenCV/transformers — the full default
# CUDA torch wheel alone is ~530MB, plus the entire NVIDIA CUDA toolkit
# dependency chain — for code neither the api container nor a CI test run
# ever calls.
#
# The degraded text-only fallback (ALLOW_PYMUPDF_FALLBACK) comes from
# pymupdf, which IS here.
#
# Keep in sync with requirements.txt when adding non-MinerU dependencies.

# Web framework & server
fastapi[standard]
uvicorn[standard]
python-multipart

# Data validation & settings
pydantic>=2.0
pydantic-settings

# HTTP client (Ollama / cloud LLM APIs)
httpx

# Database
sqlalchemy[asyncio]>=2.0
asyncpg
psycopg2-binary       # sync driver used by Celery workers
pgvector

# File & PDF handling
aiofiles
pypdf
pymupdf               # page rasterisation for the VLM extractor + text fallback

# Web article extraction (services/article_extraction.py) — architecture-
# independent, needed regardless of which PDF extractor provider is active.
trafilatura==2.2.0

# Auth: Argon2id password hashing (OWASP-recommended default) + email
# validation for pydantic's EmailStr (fastapi[standard] usually pulls this in
# transitively, but pinning it directly avoids depending on that staying true).
argon2-cffi>=23.1
email-validator>=2.0

# Background processing (redis>=5.0 also backs sessions via redis.asyncio — see app/core/redis.py)
celery>=5.3
redis>=5.0
